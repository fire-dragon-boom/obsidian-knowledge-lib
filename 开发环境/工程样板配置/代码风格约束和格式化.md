## npm配置

```bash
# 核心包
npm i -D eslint

# ts-eslint插件配置
npm i -D typescript-eslint

# prettier配置
npm i -D prettier eslint-config-prettier

# node.js
npm i -D @eslint/js
```

```json
//package.json
"devDependencies": {
	"@eslint/js": "^10.0.1",
	"eslint": "^10.4.0",
	"eslint-config-prettier": "^10.1.8",
	"prettier": "^3.8.3",
	"typescript-eslint": "^8.59.3",
}
```

## prettier

### .prettierrc.js
```js
// .prettierrc.js

/**

 * Prettier 配置文件

 * 文档：https://prettier.io/docs/en/options.html

 */

  

/** @type {import('prettier').Config} */

module.exports = {

  // ===== 基础格式 =====

  /**

   * 每行最大宽度（字符数）

   * 默认：80

   * 超过这个长度会自动换行

   * 建议：团队习惯用 80 或 100

   */

  printWidth: 100,

  /**

   * 缩进空格数

   * 默认：2

   * 注意：Prettier 不支持 Tab，只支持空格

   */

  tabWidth: 2,

  /**

   * 使用空格而不是 Tab

   * 默认：true

   * 几乎总是保持 true

   */

  useTabs: false,

  /**

   * 语句末尾加分号

   * 默认：true

   * true: 总是加分号 const a = 1;

   * false: 只在需要时加分号 const a = 1

   */

  semi: true,

  /**

   * 使用单引号而不是双引号

   * 默认：false (双引号)

   * true: 单引号 'hello'

   * false: 双引号 "hello"

   */

  singleQuote: true,

  /**

   * JSX 中使用单引号还是双引号

   * 默认：false (双引号)

   * 建议：通常和 singleQuote 保持一致

   */

  jsxSingleQuote: false,

  /**

   * 多行时是否加尾随逗号

   * 默认：'es5' (ES5 中有效的尾随逗号)

   * 可选值：

   *   - 'none': 不加逗号

   *   - 'es5': 在 ES5 中有效的地方加（对象、数组等）

   *   - 'all': 所有可能的地方加（函数参数等）

   */

  trailingComma: 'es5',

  /**

   * 对象字面量中括号之间的空格

   * 默认：true

   * true: { foo: bar }

   * false: {foo: bar}

   */

  bracketSpacing: true,

  /**

   * JSX 中 > 是否放在最后一行的末尾

   * 默认：false

   * false: <button className="btn">

   *          Click me

   *        </button>

   * true:  <button className="btn">

   *          Click me

   *        </button>

   */

  bracketSameLine: false,

  /**

   * 箭头函数参数是否始终加括号

   * 默认：'always'

   * 可选值：

   *   - 'always': 总是加括号 (x) => x

   *   - 'avoid': 尽可能省略括号 x => x

   */

  arrowParens: 'always',

  // ===== Vue / 模板相关 =====

  /**

   * Vue 文件中缩进 script 和 style 标签内的代码

   * 默认：false

   * true: <script>

   *         console.log('indented')

   *       </script>

   */

  vueIndentScriptAndStyle: false,

  /**

   * HTML 中空白符的敏感度

   * 默认：'css'

   * 可选值：

   *   - 'css': 遵循 CSS display 规则

   *   - 'strict': 所有空白符都敏感

   *   - 'ignore': 忽略空白符

   */

  htmlWhitespaceSensitivity: 'css',

  /**

   * 是否在文件末尾添加换行符

   * 默认：true

   * 建议：true（Unix 规范）

   */

  endOfLine: 'lf',

  /**

   * 是否格式化嵌入的内容（如 Markdown 中的代码块）

   * 默认：'auto'

   * 'auto': 识别标记为 prettier 或格式化后的代码

   */

  embeddedLanguageFormatting: 'auto',

  /**

   * HTML、Vue、JSX 中每行的属性数量

   * 默认：1 (每个属性单独一行)

   * 注意：这个选项在较新版本中可能被废弃

   */

  // singleAttributePerLine: false,

  // ===== 插件（可选）=====

  /**

   * 插件列表

   * 例如：需要格式化 PHP、Java 等非 JS 语言时

   */

  // plugins: ['@prettier/plugin-php'],

  // ===== 覆盖特定文件的配置 =====

  /**

   * 针对不同文件类型覆盖默认配置

   */

  overrides: [

    {

      files: '*.json',

      options: {

        singleQuote: false,  // JSON 必须用双引号

        trailingComma: 'none', // JSON 不允许尾随逗号

      },

    },

    {

      files: '*.md',

      options: {

        printWidth: 80,      // Markdown 文件使用更窄的宽度

        proseWrap: 'always', // 对 prose 文本进行换行

      },

    },

    {

      files: '*.{yaml,yml}',

      options: {

        singleQuote: false,

        tabWidth: 2,

      },

    },

  ],

};


```

## eslint
### eslint.config.js


```js
// eslint.config.js

import js from '@eslint/js';

import tseslint from 'typescript-eslint';

import eslintConfigPrettier from 'eslint-config-prettier';

  

/** @type {import('eslint').Linter.FlatConfig[]} */

export default [

  // JS 推荐规则

  js.configs.recommended,

  

  // TS 推荐规则

  ...tseslint.configs.recommended,

  

  // 关闭与 prettier 冲突的规则（必须放最后）

  eslintConfigPrettier,

  

  // 自定义规则

  {

    files: ['**/*.ts'],

    languageOptions: {

      parserOptions: {

        project: true,

      },

    },

  

    rules: {

      'no-console': 'off',

      'no-unused-vars': 'off',

      '@typescript-eslint/no-unused-vars': [

        'error',

        {

          args: 'all',

          argsIgnorePattern: '^_',

          caughtErrors: 'all',

          caughtErrorsIgnorePattern: '^_',

          destructuredArrayIgnorePattern: '^_',

          varsIgnorePattern: '^_',

          ignoreRestSiblings: true,

        },

      ],

    },

  },

];
```
